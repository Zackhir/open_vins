\section{Protocol Summary}
%==============================================================================

\begin{center}
\begin{tabular}{@{}>{\bfseries\color{brand}}l l@{}}
\toprule
Estimator & OpenVINS MSCKF (\texttt{serial.launch}) \\
Camera mode & Mono: \texttt{max\_cameras:=1}, \texttt{use\_stereo:=false} (left / cam0) \\
Alignment & \texttt{posyaw} via \texttt{ov\_eval error\_singlerun} \\
Table metrics & Trans.\ RMSE (m), Ori.\ RMSE (deg), ms/frame, FPS, CPU mean/max \% \\
Error plots & \texttt{error\_dataset}-style RMSE (orientation + position vs time) \\
Trajectories & XY and Z vs ground truth \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.6em}
\begin{tcolorbox}[colframe=warn, colback=softbg]
\textbf{Important rules}
\begin{itemize}[leftmargin=1.2em, itemsep=0.2em]
  \item Do \textbf{not} run a separate \texttt{rosbag play} with \texttt{serial.launch} --- the serial node already reads the bag.
  \item Do \textbf{not} average log \texttt{(xx hz)} lines for FPS --- that is print rate, not processing rate.
  \item Correct FPS: \(1 / \mathrm{mean\_total}\) from \texttt{timing.txt} (requires \texttt{dotime:=true}).
  \item CPU: mean/max \% of process \texttt{ros1\_serial\_msckf} (100\% \(\approx\) one full core).
\end{itemize}
\end{tcolorbox}

\section{Manual Single-Sequence Example}
%==============================================================================

\subsection{Terminal 1 --- ROS master}
\begin{lstlisting}[language=bash]
ov_docker ov_ros1_20_04 bash
source /opt/ros/noetic/setup.bash
source /catkin_ws/devel/setup.bash
roscore
\end{lstlisting}

\subsection{Terminal 2 --- EuRoC MH\_01 (mono)}
\begin{lstlisting}[language=bash]
ov_docker ov_ros1_20_04 bash
source /opt/ros/noetic/setup.bash
source /catkin_ws/devel/setup.bash

SEQ=MH_01_easy
OUT=/results/baseline_mono_msckf/${SEQ}
mkdir -p "$OUT"

roslaunch ov_msckf serial.launch \
  config:=euroc_mav \
  dataset:=${SEQ} \
  bag:=/datasets/machine_hall/${SEQ}/${SEQ}.bag \
  bag_start:=40 \
  max_cameras:=1 \
  use_stereo:=false \
  dosave:=true \
  dotime:=true \
  path_est:=${OUT}/estimate.txt \
  path_time:=${OUT}/timing.txt
\end{lstlisting}

\subsection{Accuracy + timing after the run}
\begin{lstlisting}[language=bash]
GT=/datasets/machine_hall/${SEQ}/${SEQ}/mav0/state_groundtruth_estimate0/data.txt
rosrun ov_eval error_singlerun posyaw "$GT" ${OUT}/estimate.txt
rosrun ov_eval plot_trajectories posyaw "$GT" ${OUT}/estimate.txt
rosrun ov_eval timing_flamegraph ${OUT}/timing.txt
# ms/frame = mean_time(total) * 1000
# FPS     = 1 / mean_time(total)
\end{lstlisting}

\subsection{UZH outdoor forward\_1}

\section{Metrics Definitions}
%==============================================================================

\begin{center}
\begin{tabular}{@{}p{3.2cm}p{11.5cm}@{}}
\toprule
\textbf{Metric} & \textbf{Meaning / source} \\
\midrule
Trans.\ RMSE & Absolute trajectory translation RMSE after \texttt{posyaw} alignment (\texttt{error\_singlerun}) \\
Ori.\ RMSE & Absolute trajectory orientation RMSE (deg), same tool \\
ms/frame & Mean of last column (\texttt{total}) in \texttt{timing.txt} \(\times 1000\) \\
FPS & \(1 / \mathrm{mean\_total}\) --- offline serial throughput, not camera rate \\
CPU mean/max & psutil sample of \texttt{ros1\_serial\_msckf} during the run \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.5em}
\noindent RPE columns are \textbf{not} required for this deliverable and are omitted from the report table.
